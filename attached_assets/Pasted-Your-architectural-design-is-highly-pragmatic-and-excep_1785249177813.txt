Hydrate the New Layout SequentiallyTake your flat array of 11 players and distribute them directly into the new formation's structural buckets using simple array slicing.javascriptfunction resizeSlots(currentSlots, newFormation) {
  // 1. Flatten
  const players = [
    ...(currentSlots.GK || []),
    ...(currentSlots.DEF || []),
    ...(currentSlots.MID || []),
    ...(currentSlots.FWD || [])
  ];

  // 2. Look up the counts needed for the target formation
  // E.g., for "3-5-2": { GK: 1, DEF: 3, MID: 5, FWD: 2 }
  const counts = {
    GK: FORMATION_SLOT_ORDER[newFormation].GK.length,
    DEF: FORMATION_SLOT_ORDER[newFormation].DEF.length,
    MID: FORMATION_SLOT_ORDER[newFormation].MID.length,
    FWD: FORMATION_SLOT_ORDER[newFormation].FWD.length,
  };

  // 3. Hydrate cleanly from left-to-right
  return {
    GK:  players.slice(0, counts.GK),
    DEF: players.slice(counts.GK, counts.GK + counts.DEF),
    MID: players.slice(counts.GK + counts.DEF, counts.GK + counts.DEF + counts.MID),
    FWD: players.slice(counts.GK + counts.DEF + counts.MID)
  };
}
